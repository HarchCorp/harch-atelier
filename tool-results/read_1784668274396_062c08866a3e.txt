    20→function mulberry32(seed: number) {
    21→  let a = seed >>> 0;
    22→  return function () {
    23→    a = (a + 0x6d2b79f5) >>> 0;
    24→    let t = a;
    25→    t = Math.imul(t ^ (t >>> 15), t | 1);
    26→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    27→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    28→  };
    29→}
    30→const rnd = mulberry32(20251115);
    31→
    32→function pick<T>(arr: readonly T[]): T {
    33→  return arr[Math.floor(rnd() * arr.length)];
    34→}

   282→function buildLatencySeries(base: number, jitter: number, count = 24): number[] {
   283→  const out: number[] = [];
   284→  for (let i = 0; i < count; i++) {
   285→    out.push(Math.max(20, Math.round(base + (rnd() - 0.5) * 2 * jitter)));
   286→  }
   287→  return out;

  1031→    const d = new Date("2025-11-15");
  1032→    d.setDate(d.getDate() - i);
  1033→    const total = Math.round(40 + 30 * Math.sin(i / 2.3) + (rnd() - 0.5) * 12);
  1034→    const failure = Math.round(total * (0.04 + rnd() * 0.08));
  1035→    out.push({
  1036→      day: d.toISOString().slice(5, 10),
  1037→      success: total - failure,
  1038→      failure,
  1039→    });
  1040→  }
  1041→  return out;
  1042→})();

  1060→export function relativeTime(iso: string, now = "2025-11-15T10:30:00Z"): string {
  1061→  const diffMs = Date.parse(now) - Date.parse(iso);
  1062→  const mins = Math.round(diffMs / 60000);
  1063→  if (mins < 1) return "just now";
  1064→  if (mins < 60) return `${mins}m ago`;
  1065→  const hrs = Math.round(mins / 60);
  1066→  if (hrs < 24) return `${hrs}h ago`;
  1067→  const days = Math.round(hrs / 24);
  1068→  if (days < 30) return `${days}d ago`;
  1069→  return new Date(iso).toISOString().slice(0, 10);
  1070→}
