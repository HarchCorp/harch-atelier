   130→  const d = new Date(2025, 0, 31); // anchored reference date for deterministic output

   354→function buildArticlesForEvent(ev: RiskEvent, count: number): Article[] {
   355→  const heads = headlineBank[ev.pillar];
   356→  return Array.from({ length: count }, (_, i) => {
   357→    const outlet = outlets[(i + ev.title.length) % outlets.length];
   358→    const head = heads[(i + ev.id.length) % heads.length];
   359→    const daysAgo = i % 6;
   360→    const d = new Date(2025, 0, 29 - daysAgo, 8 + (i % 10), (i * 7) % 60);
   361→    const sentimentRoll = (i + ev.articles) % 10;
   362→    const sentiment: Sentiment =
   363→      ev.sentiment === "negative"
   364→        ? sentimentRoll < 7
   365→          ? "negative"
   366→          : sentimentRoll < 9
   367→            ? "neutral"
   368→            : "positive"
   369→        : ev.sentiment === "positive"
   370→          ? sentimentRoll < 7
   371→            ? "positive"
   372→            : "neutral"
   373→          : "neutral";
   374→    return {
   375→      id: `ART-${ev.id.replace("EVT-", "")}-${String(i + 1).padStart(3, "0")}`,
   376→      eventId: ev.id,
   377→      headline: head,
   378→      source: outlet.name,
   379→      tier: outlet.tier,
   380→      url: "#",
   381→      publishedAt: d.toISOString(),
   382→      sentiment,
   383→      reach: outlet.reach - (i % 5) * 12000,
   384→      language: languages[(i + ev.pillar.length) % languages.length],
   385→    };
   386→  });
   387→}

   452→        ? "neutral"
   453→        : "positive";

   457→    date: d.toISOString().slice(0, 10),
   458→    pillar: p.pillar,
   459→    title: p.name,
   460→    articles: p.articles,
   461→    sentiment,
   462→    severity: p.severity,
   463→    region: p.region,
   464→  };
   465→}

   468→/*  Risk pillar aggregation (for the new RiskPillars widget)           */
   469→/* ------------------------------------------------------------------ */
   470→
   471→export interface PillarAgg {
   472→  pillar: RiskPillar;
   473→  events: number;
   474→  articles: number;
   475→  /** weighted 0–100 exposure score. */
   476→  exposure: number;
   477→  /** -100..+100 net sentiment skew. */
   478→  sentimentSkew: number;
   479→}
   480→
   481→export const pillarAgg: PillarAgg[] = (() => {
   482→  const map = new Map<RiskPillar, PillarAgg>();
   483→  for (const p of [
   484→    "Regulatory",
   485→    "Cyber",
   486→    "Financial",
   487→    "ESG",
   488→    "Geopolitical",
   489→    "Reputational",
   490→  ] as RiskPillar[]) {
   491→    map.set(p, { pillar: p, events: 0, articles: 0, exposure: 0, sentimentSkew: 0 });
   492→  }
   493→  for (const e of riskEvents) {
   494→    const row = map.get(e.pillar)!;
   495→    row.events += 1;
   496→    row.articles += e.articles;
   497→    row.sentimentSkew +=
   498→      e.sentiment === "negative" ? -1 : e.sentiment === "positive" ? 1 : 0;
   499→  }
   500→  const maxArticles = Math.max(...[...map.values()].map((r) => r.articles), 1);
   501→  for (const row of map.values()) {
   502→    // exposure = weighted blend of volume + severity-weighted event count
   503→    const criticalHits = riskEvents.filter(
   504→      (e) => e.pillar === row.pillar && (e.severity === "critical" || e.severity === "high"),
   505→    ).length;
   506→    row.exposure = Math.min(
   507→      100,
   508→      Math.round((row.articles / maxArticles) * 60 + criticalHits * 12 + row.events * 3),
   509→    );
   510→    row.sentimentSkew = Math.max(-100, Math.min(100, Math.round((row.sentimentSkew / Math.max(row.events, 1)) * 100)));
   511→  }
   512→  return [...map.values()].sort((a, b) => b.exposure - a.exposure);
   513→})();

   540→/* ------------------------------------------------------------------ */
   541→/*  Time-range slicing helpers for Media Coverage                      */
   542→/* ------------------------------------------------------------------ */
   543→
   544→export type CoverageRange = "7d" | "30d" | "90d";
   545→
   546→export function sliceCoverage(range: CoverageRange): CoverageDay[] {
   547→  if (range === "30d") return coverage30d;
   548→  if (range === "7d") return coverage30d.slice(-7);
   549→  // 90d — synthesize by repeating the 30d pattern twice more with a slight drift
   550→  const base = coverage30d;
   551→  const extended: CoverageDay[] = [];
   552→  for (let cycle = 2; cycle >= 0; cycle--) {
   553→    for (let i = 0; i < base.length; i++) {
   554→      const drift = cycle * 0.85;
   555→      extended.push({
   556→        date: base[i].date,
   557→        positive: Math.round(base[i].positive * drift + (cycle * 4)),
   558→        negative: Math.round(base[i].negative * drift + (cycle * 3)),
   559→      });
   560→    }
   561→  }
   562→  // re-anchor dates so the 90d window ends "today"
   563→  const today = new Date(2025, 0, 31);
   564→  return extended.map((d, idx) => {
   565→    const dt = new Date(today);
   566→    dt.setDate(dt.getDate() - (extended.length - 1 - idx));
   567→    return { ...d, date: dt.toISOString().slice(0, 10) };
   568→  });
   569→}

   626→/*  Risk trend timeline (90d risk-index with peak markers)            */
   627→/* ------------------------------------------------------------------ */
   628→
   629→export interface RiskTrendPoint {
   630→  /** day offset from the start of the 90d window (0 = oldest). */
   631→  day: number;
   632→  /** ISO date short. */
   633→  date: string;
   634→  /** composite risk index 0–100. */
   635→  index: number;
   636→  /** short label for any annotated peak (undefined if none). */
   637→  peak?: string;
   638→}
   639→
   640→function buildRiskTrend90d(): RiskTrendPoint[] {
   641→  const points: RiskTrendPoint[] = [];
   642→  const today = new Date();
   643→  for (let i = 89; i >= 0; i--) {
   644→    const d = new Date(today);
   645→    d.setDate(d.getDate() - i);
   646→    const t = 89 - i; // 0..89
   647→    // Smooth pseudo-random walk centered around 65 with occasional spikes.
   648→    const base = 62 + 8 * Math.sin(t / 7.5) + 6 * Math.cos(t / 13.2);
   649→    const noise = ((t * 9301 + 49297) % 233280) / 233280 * 8 - 4;
   650→    const index = Math.max(18, Math.min(96, Math.round(base + noise)));
   651→    const point: RiskTrendPoint = {
   652→      day: 89 - i,
   653→      date: d.toISOString().slice(0, 10),
   654→      index,
   655→    };
   656→    // Mark the 3 highest peaks.
   657→    if (t === 17) point.peak = "SEC inquiry leak";
   658→    if (t === 44) point.peak = "Ransomware claim";
   659→    if (t === 71) point.peak = "Analyst downgrade";
   660→    points.push(point);
   661→  }
   662→  return points;
   663→}
