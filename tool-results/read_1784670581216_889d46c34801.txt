   131→/** Mulberry32 — tiny deterministic PRNG so mock data is stable per build. */
   132→function mulberry32(seed: number): () => number {
   133→  let a = seed >>> 0;
   134→  return () => {
   135→    a = (a + 0x6d2b79f5) >>> 0;
   136→    let t = a;
   137→    t = Math.imul(t ^ (t >>> 15), t | 1);
   138→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
   139→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   140→  };
   141→}

   143→/** Build a 30-trading-day series anchored at a target close. */
   144→function buildSeries30d(seed: number, endPrice: number, dailyVolPct: number): number[] {
   145→  const rng = mulberry32(seed);
   146→  const out: number[] = [];
   147→  // Walk backwards from endPrice to day 0 with mean-reverting noise.
   148→  let p = endPrice;
   149→  for (let i = 0; i < 30; i++) {
   150→    out.unshift(p);
   151→    const shock = (rng() - 0.5) * 2 * (p * dailyVolPct) / 100;
   152→    p = Math.max(0.5, p - shock);
   153→  }
   154→  return out;
   155→}

   157→function isoDaysAgo(days: number): string {
   158→  const d = new Date();
   159→  d.setDate(d.getDate() - days);
   160→  return d.toISOString().slice(0, 10);
   161→}

   163→/* ------------------------------------------------------------------ */
   164→/*  MASI index — 30d + intraday                                       */
   165→/* ------------------------------------------------------------------ */
   166→
   167→function buildMasi30d(): IndexPoint[] {
   168→  const rng = mulberry32(20250117);
   169→  const out: IndexPoint[] = [];
   170→  let masi = 13850;
   171→  let masi20 = 1455;
   172→  for (let i = 29; i >= 0; i--) {
   173→    const date = isoDaysAgo(i);
   174→    masi = masi + (rng() - 0.45) * 35 + Math.sin(i / 4) * 6;
   175→    masi20 = masi20 + (rng() - 0.45) * 4 + Math.sin(i / 5) * 1.2;
   176→    out.push({
   177→      date,
   178→      masi: Math.round(masi * 100) / 100,
   179→      masi20: Math.round(masi20 * 100) / 100,
   180→    });
   181→  }
   182→  return out;
   183→}

   191→function buildMasiIntraday(): IntradayPoint[] {
   192→  const rng = mulberry32(98765);
   193→  const out: IntradayPoint[] = [];
   194→  // BVC trading session: 09:00 – 17:00 with 15-min ticks.
   195→  const baseMasi = masiLatest.masi - 12;
   196→  const baseMasi20 = masiLatest.masi20 - 1.5;
   197→  for (let h = 9; h < 17; h++) {
   198→    for (const m of [0, 15, 30, 45]) {
   199→      if (h === 9 && m === 0) continue;
   200→      const label = `${String(h).padStart(2, "0")}:${String(m).padStart(2, "0")}`;
   201→      const t = (h - 9) * 4 + m / 15;
   202→      const drift = Math.sin(t / 6) * 4 + (rng() - 0.5) * 9;
   203→      const turnover = 120 + (rng() * 90) + (t > 16 ? 80 : 0);
   204→      out.push({
   205→        time: label,
   206→        masi: Math.round((baseMasi + drift) * 100) / 100,
   207→        masi20: Math.round((baseMasi20 + drift / 8) * 100) / 100,
   208→        volumeM: Math.round(turnover),
   209→      });
   210→    }
   211→  }
   212→  return out;
   213→}

   215→export const masiIntraday: IntradayPoint[] = buildMasiIntraday();

   233→    high: Math.round(high * 100) / 100,

   235→    open: Math.round(open * 100) / 100,
   236→    last: Math.round(last * 100) / 100,
   237→    chgFromOpen: Math.round((last - open) * 100) / 100,

   412→/** Compute M2M P&L for a position (long: (last-avg)*qty; short: (avg-last)*qty). */
   413→export function positionPnl(p: Position): number {
   414→  const dir = p.side === "long" ? 1 : -1;
   415→  return Math.round((p.last - p.avgPrice) * p.qty * dir * 100) / 100;
   416→}
   417→
   418→export function positionPnlPct(p: Position): number {
   419→  const dir = p.side === "long" ? 1 : -1;
   420→  return Math.round(((p.last - p.avgPrice) * dir / p.avgPrice) * 10000) / 100;
   421→}
   422→
   423→export function positionExposure(p: Position): number {
   424→  return Math.round(p.qty * p.last * 100) / 100;
   425→}
   426→
   427→export function positionCost(p: Position): number {
   428→  return Math.round(p.qty * p.avgPrice * 100) / 100;
   429→}

   431→/** Aggregate position summary. */
   432→export const positionsSummary = (() => {
   433→  const totalExposure = positions.reduce((s, p) => s + positionExposure(p), 0);
   434→  const totalCost = positions.reduce((s, p) => s + positionCost(p), 0);
   435→  const totalPnl = positions.reduce((s, p) => s + positionPnl(p), 0);
   436→  const longExposure = positions
   437→    .filter((p) => p.side === "long")
   438→    .reduce((s, p) => s + positionExposure(p), 0);
   439→  const shortExposure = positions
   440→    .filter((p) => p.side === "short")
   441→    .reduce((s, p) => s + positionExposure(p), 0);
   442→  const winners = positions.filter((p) => positionPnl(p) > 0).length;
   443→  const losers = positions.filter((p) => positionPnl(p) <= 0).length;
   444→  return {
   445→    totalExposure: Math.round(totalExposure),
   446→    totalCost: Math.round(totalCost),
   447→    totalPnl: Math.round(totalPnl),
   448→    totalPnlPct: Math.round((totalPnl / totalCost) * 10000) / 100,
   449→    longExposure: Math.round(longExposure),
   450→    shortExposure: Math.round(shortExposure),
   451→    longPct: Math.round((longExposure / (longExposure + shortExposure)) * 1000) / 10,
   452→    shortPct: Math.round((shortExposure / (longExposure + shortExposure)) * 1000) / 10,
   453→    winners,
   454→    losers,
   455→    count: positions.length,
   456→  };
   457→})();

   474→  }
   475→  const total = [...map.values()].reduce((s, v) => s + v.exposure, 0);
   476→  return [...map.entries()]
   477→    .map(([sector, v]) => ({
   478→      sector,
   479→      exposure: Math.round(v.exposure),
   480→      pnl: Math.round(v.pnl),
   481→      pct: Math.round((v.exposure / total) * 1000) / 10,
   482→    }))
   483→    .sort((a, b) => b.exposure - a.exposure);
   484→})();
   485→
   486→/* ------------------------------------------------------------------ */
   487→/*  MAD FX (EUR/MAD, USD/MAD, GBP/MAD) — 30d                          */
   488→/* ------------------------------------------------------------------ */
   489→
   490→function buildFx30d(): FxPoint[] {
   491→  const rng = mulberry32(424242);
   492→  const out: FxPoint[] = [];
   493→  let eur = 10.82;
   494→  let usd = 10.02;
   495→  let gbp = 12.65;
   496→  for (let i = 29; i >= 0; i--) {
   497→    eur = eur + (rng() - 0.48) * 0.025 + Math.sin(i / 5) * 0.008;
   498→    usd = usd + (rng() - 0.5) * 0.03 + Math.sin(i / 4) * 0.012;
   499→    gbp = gbp + (rng() - 0.47) * 0.04 + Math.sin(i / 6) * 0.015;
   500→    out.push({
   501→      date: isoDaysAgo(i),
   502→      eurMad: Math.round(eur * 10000) / 10000,
   503→      usdMad: Math.round(usd * 10000) / 10000,
   504→      gbpMad: Math.round(gbp * 10000) / 10000,
   505→    });
   506→  }
   507→  return out;
   508→}

   651→export function formatCompactMAD(n: number): string {

   653→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}M`;
   654→  return `${Math.round(n)}`;

   660→  return `${n}`;

   675→/** Heat color for sector heatmap tiles (red-green diverging). */
   676→export function heatColor(chgPct: number, max: number = 1.0): string {
   677→  const t = Math.max(-1, Math.min(1, chgPct / (max * 1.2)));
   678→  if (t > 0) {
   679→    const alpha = 0.18 + t * 0.55;
   680→    return `rgba(16, 185, 129, ${alpha.toFixed(2)})`;
   681→  } else if (t < 0) {
   682→    const alpha = 0.18 + Math.abs(t) * 0.55;
   683→    return `rgba(225, 29, 72, ${alpha.toFixed(2)})`;
   684→  }
   685→  return "rgba(148, 163, 184, 0.12)";
   686→}
