    86→/* ------------------------------------------------------------------ */
    87→
    88→function Sparkline({
    89→  data,
    90→  color,
    91→  width = 96,
    92→  height = 28,
    93→}: {
    94→  data: number[];
    95→  color: string;
    96→  width?: number;
    97→  height?: number;
    98→}) {
    99→  const id = React.useId();
   100→  const max = Math.max(...data);
   101→  const min = Math.min(...data);
   102→  const range = max - min || 1;
   103→  const step = width / (data.length - 1);
   104→  const pts = data.map(
   105→    (v, i) => `${(i * step).toFixed(1)},${(height - ((v - min) / range) * height).toFixed(1)}`,
   106→  );
   107→  const areaPts = `0,${height} ${pts.join(" ")} ${width},${height}`;
   108→  return (
   109→    <svg
   110→      width={width}
   111→      height={height}
   112→      viewBox={`0 0 ${width} ${height}`}
   113→      className="overflow-visible"
   114→      preserveAspectRatio="none"
   115→      aria-hidden
   116→    >
   117→      <defs>
   118→        <linearGradient id={id} x1="0" y1="0" x2="0" y2="1">
   119→          <stop offset="0%" stopColor={color} stopOpacity={0.28} />
   120→          <stop offset="100%" stopColor={color} stopOpacity={0} />
   121→        </linearGradient>
   122→      </defs>
   123→      <polygon points={areaPts} fill={`url(#${id})`} />
   124→      <polyline
   125→        points={pts.join(" ")}
   126→        fill="none"
   127→        stroke={color}
   128→        strokeWidth={1.5}
   129→        strokeLinejoin="round"
   130→        strokeLinecap="round"
   131→      />
   132→    </svg>
   133→  );
   134→}

   136→/* ------------------------------------------------------------------ */
   137→/*  Risk ring (mini circular gauge, 0–100)                             */
   138→/* ------------------------------------------------------------------ */
   139→
   140→function RiskRing({ score, size = 44 }: { score: number; size?: number }) {
   141→  const stroke = 4;
   142→  const r = (size - stroke) / 2;
   143→  const c = 2 * Math.PI * r;
   144→  const pct = Math.max(0, Math.min(100, score));
   145→  const offset = c * (1 - pct / 100);
   146→  const color = score >= 70 ? "#e11d48" : score >= 50 ? "#f59e0b" : "#10b981";
   147→  const verdict = score >= 70 ? "High" : score >= 50 ? "Mod" : "Low";
   148→  return (
   149→    <div className="relative inline-flex" style={{ width: size, height: size }}>
   150→      <svg width={size} height={size} className="-rotate-90">
   151→        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke="#f1f5f9" strokeWidth={stroke} />
   152→        <circle
   153→          cx={size / 2}
   154→          cy={size / 2}
   155→          r={r}
   156→          fill="none"
   157→          stroke={color}
   158→          strokeWidth={stroke}
   159→          strokeLinecap="round"
   160→          strokeDasharray={c}
   161→          strokeDashoffset={offset}
   162→          style={{ transition: "stroke-dashoffset 0.6s ease-out" }}
   163→        />
   164→      </svg>
   165→      <div className="pointer-events-none absolute inset-0 flex flex-col items-center justify-center">
   166→        <span className="tabular text-[11px] font-bold leading-none text-slate-900">
   167→          {score.toFixed(0)}
   168→        </span>
   169→        <span className="text-[8px] font-semibold uppercase leading-none" style={{ color }}>
   170→          {verdict}
   171→        </span>
   172→      </div>
   173→    </div>
   174→  );
   175→}

   350→  return { text: "text-rose-700", bg: "bg-rose-50" };
