   130→/* ------------------------------------------------------------------ */
   131→/*  Seeded PRNG (deterministic)                                        */
   132→/* ------------------------------------------------------------------ */
   133→
   134→function mulberry32(seed: number): () => number {
   135→  let a = seed >>> 0;
   136→  return () => {
   137→    a = (a + 0x6d2b79f5) >>> 0;
   138→    let t = a;
   139→    t = Math.imul(t ^ (t >>> 15), t | 1);
   140→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
   141→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   142→  };
   143→}
   144→
   145→const rng = mulberry32(20251118);
   146→
   147→function ri(min: number, max: number): number {
   148→  return Math.round(min + rng() * (max - min));
   149→}
   150→
   151→function rf(min: number, max: number, dp = 1): number {
   152→  const v = min + rng() * (max - min);
   153→  const f = Math.pow(10, dp);
   154→  return Math.round(v * f) / f;
   155→}
   156→
   157→function pick<T>(arr: readonly T[]): T {
   158→  return arr[Math.floor(rng() * arr.length)];
   159→}
   160→
   161→/** Build a 12-month sentiment series ending at a target net score. */
   162→function buildSentiment12m(endScore: number, vol: number): number[] {
   163→  const out: number[] = [];
   164→  let p = endScore - vol * 0.6;
   165→  for (let i = 0; i < 12; i++) {
   166→    const shock = (rng() - 0.5) * vol;
   167→    p = Math.max(-95, Math.min(95, p + shock + (endScore - p) * 0.18));
   168→    out.push(Math.round(p * 10) / 10);
   169→  }
   170→  out[11] = endScore;
   171→  return out;
   172→}
   173→
   174→/** Build a 20-point sparkline ending at a target. */
   175→function buildSparkline(end: number, volPct: number): number[] {
   176→  const out: number[] = [];
   177→  let p = end * (1 - volPct * 0.4);
   178→  for (let i = 0; i < 20; i++) {
   179→    const shock = (rng() - 0.5) * end * volPct;
   180→    p = Math.max(0.5, p + shock + (end - p) * 0.22);
   181→    out.push(Math.round(p * 100) / 100);
   182→  }
   183→  out[19] = end;
   184→  return out;
   185→}
   186→
   187→function buildPillars(seed: number, base: number): EntityPillarScore {
   188→  const r = mulberry32(seed);
   189→  const jitter = (lo: number, hi: number) => Math.round(lo + r() * (hi - lo));
   190→  return {
   191→    regulatory: Math.max(8, Math.min(96, base + jitter(-12, 12))),
   192→    cyber: Math.max(8, Math.min(96, base + jitter(-10, 14))),
   193→    financial: Math.max(8, Math.min(96, base + jitter(-14, 10))),
   194→    esg: Math.max(8, Math.min(96, base + jitter(-16, 14))),
   195→    geopolitical: Math.max(8, Math.min(96, base + jitter(-18, 16))),
   196→    reputational: Math.max(8, Math.min(96, base + jitter(-12, 12))),
   197→  };
   198→}
   199→
   200→/** Composite score from a pillar map (rounded mean). */
   201→function composite(p: EntityPillarScore): number {
   202→  const v =
   203→    (p.regulatory +
   204→      p.cyber +
   205→      p.financial +
   206→      p.esg +
   207→      p.geopolitical +
   208→      p.reputational) /
   209→    6;
   210→  return Math.round(v * 10) / 10;
   211→}

   217→export function formatMAD(n: number, dp = 2): string {
   218→  return n.toLocaleString("en-US", {
   219→    minimumFractionDigits: dp,
   220→    maximumFractionDigits: dp,
   221→  });
   222→}
   223→
   224→export function formatCompactMAD(n: number): string {
   225→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(2)}B`;
   226→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}M`;
   227→  return `${Math.round(n)}`;
   228→}
   229→
   230→export function formatNumber(n: number): string {
   231→  return n.toLocaleString("en-US");
   232→}
   233→
   234→export function formatCompact(n: number): string {
   235→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(2)}M`;
   236→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}K`;
   237→  return `${Math.round(n)}`;
   238→}

   807→/** Convert a listed Equity from market-data into a full Entity. */
   808→function listedToEntity(ticker: string): Entity {
   809→  const eq = moroccanEquities.find((e) => e.ticker === ticker);
   810→  if (!eq) throw new Error(`Equity ${ticker} not found in moroccanEquities`);
   811→  const ext = listedExtensions[ticker];
   812→  const pillars = buildPillars(200 + ticker.charCodeAt(0) + ticker.charCodeAt(1) % 256, ext.riskBase);
   813→  return {
   814→    id: ticker,
   815→    name: eq.name,
   816→    ticker,
   817→    type: "listed",
   818→    sector: eq.sector,
   819→    hq: ext.hq,
   820→    country: "Morocco",
   821→    employees: ext.employees,
   822→    revenueM: ext.revenueM,
   823→    netIncomeM: ext.netIncomeM,
   824→    assetsM: ext.assetsM,
   825→    mktCapM: eq.mktCapM,
   826→    peRatio: eq.peRatio,
   827→    dividendYield: ext.dividendYield,
   828→    riskScore: composite(pillars),
   829→    riskPillars: pillars,
   830→    sentiment: ext.sentiment,
   831→    sentimentTrend12m: buildSentiment12m(ext.sentiment, 18),
   832→    lastNews: ext.lastNews,
   833→    leadership: ext.leadership,
   834→    ownership: ext.ownership,
   835→    status: ext.status,
   836→    watchlisted: ext.watchlisted,
   837→    sparkline: buildSparkline(eq.last, 0.04),
   838→  };
   839→}
   840→
   841→const listedEntities: Entity[] = moroccanEquities.map((eq) => listedToEntity(eq.ticker));
